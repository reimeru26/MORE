 19def calcYear(w, offset, years):
 20    # this is for fueling of a hydrogen car
 21    # we have seven days - we need a full year, one year has 52-53 weeks
 22    w = w * 53 * years
 23    d = w[offset:] # this is days
 24    
 25    #now, create hours
 26    hours = []
 27    for day in d:
 28        
 29        x = 1
 30        while x < 8:    
 31            hours.append(0.0)
 32            x = x + 1
 33        
 34        hours.append(day)   # -> fueling (in kg) at 09:00
 35        
 36        while x < 24:    
 37            hours.append(0.0)
 38            x = x + 1
 39    
 40    #print(len(hours))
 41    return hours

 43def calcRange(myres, dataCar):
 44    # calculate driving range as indication for tank status in car    
 45    # myres[7] -> contains only filling events (status THIS IS)
 46    # dataCar  -> contains filling plan (status SHOULD BE)
 47    myrange = []
 48    r = 0.0
 49    a = 0
 50    while a < len(myres[7]):
 51        
 52        if dataCar[a] > 0.0:
 53            r = 100.0 * myres[7][a] / dataCar[a]
 54            
 55        myrange.append(r)
 56        a = a + 1
 57        
 58    myres.append(myrange)
 59    return None

 62def hydrogenTank(dataCar, myIn, myres):
 63    tank = []   # hydrogen tank stationary
 64    car = []    # car tank 
 65    diff = []   # H2 overproduction (cannot be stored)
 66    hdemand = []    # heat demand that cannot be delivered by H2 (empty storage)
 67    maxStore = myIn['system'][6]
 68    stor = 0.0  # tank stationary
 69    onefill = 0.0 # car tank 
 70    
 71    hmass = myres[4] # produced H2 by electrolysis / kg
 72    hcon = myres[5]  # demand H2 by heating / kg
 73    
 74    a = 0
 75    while a < len(hmass):
 76        
 77        if hcon[a] > 0.0:   # if heating, then no H2 production
 78                stor = stor - hcon[a]        # burning H2 from tank
 79        else:
 80            stor = stor + hmass[a]       # H2 production to tank
 81            
 82        
 83        if stor < 0.0:
 84            e = stor * (-1.0)               # mass H2 in kg (which would be needed additionally)
 85            e = e * myIn['operation'][3]    # equivalent heat from combustion H2
 86            hdemand.append(e)               # additional demand in kWh
 87            stor = 0.0
 88        else:
 89            hdemand.append(0.0)
 90            
 91        onefill = 0.0
 92        if dataCar[a] > 0.0:         # car fueling
 93            if stor == 0.0:             # is empty, no filling
 94                onefill = 0.0
 95            elif stor < dataCar[a]:
 96                onefill = stor
 97            else:
 98                onefill = dataCar[a]
 99            
100            stor = stor - onefill
101        
102        #### 3) check capacity (overproduction)
103        mydiff = 0.0
104        if stor > maxStore:        
105            mydiff = stor - maxStore
106        stor = stor - mydiff
107                
108        car.append(onefill)
109        tank.append(stor)
110        diff.append(mydiff)
111        
112        a = a + 1
113       
114    myres.append(tank)
115    myres.append(car)
116    myres.append(diff)
117    myres.append(hdemand)
118    
119    return None

175def heatDemand(dataTemp, myIn, years):
176    # Calculate heat demand per hour
177    m = myIn['heating'][0]       # m2 area for heating
178    heatyear = myIn['heating'][1] # kWh / m2
179    tlim = myIn['heating'][2]
180    tnorm = myIn['heating'][3]    
181    
182    heat = []   # heat demand
183    dt = []     # temp difference inside / outside
184    sumt = 0.0
185    for a in dataTemp:
186        if a < tlim:
187            b = tnorm - a
188            dt.append(b)
189            sumt = sumt + b
190        else:
191            dt.append(0.0)
192        
193    # heat coefficient 
194    k = ( heatyear * years * m ) / sumt
195    
196    for a in dt:
197        b = k * a
198        heat.append(b)
199    
200    return heat   # heat demand in kW
